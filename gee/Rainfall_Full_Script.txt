// Faisalabad Rainfall CHIRPS 2001-2024
var roi = ee.FeatureCollection("FAO/GAUL/2015/level2").filter(ee.Filter.eq('ADM2_NAME','Faisalabad'));

var chirps = ee.ImageCollection('UCSB-CHG/CHIRPS/DAILY')
  .filterBounds(roi)
  .filterDate('2001-01-01','2024-12-31')
  .select('precipitation');

var monthlyRain = ee.List.sequence(2001, 2024).iterate(function(y, prev){
  var yearCol = ee.List.sequence(1,12).map(function(m){
    var start = ee.Date.fromYMD(y,m,1);
    var filtered = chirps.filterDate(start, start.advance(1,'month')).sum();
    var mean = filtered.reduceRegion({reducer: ee.Reducer.mean(), geometry: roi, scale: 5566, maxPixels: 1e13}).get('precipitation');
    return ee.Feature(null, {'year': y, 'month': m, 'Rainfall_mm': mean});
  });
  return ee.List(prev).cat(yearCol);
}, ee.List([]));

Export.table.toDrive({
  collection: ee.FeatureCollection(ee.List(monthlyRain)),
  description: 'Full_Historical_Rainfall_2001-2024',
  fileFormat: 'CSV',
  selectors: ['year','month','Rainfall_mm']
});
